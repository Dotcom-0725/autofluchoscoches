function roleForPassword_(password) {
  return ADMIN_ROLES[password] || null;
}

function doGet(e) {
  const action = e.parameter.action;

  if (action === 'admin') {
    const role = roleForPassword_(e.parameter.password);
    if (!role) {
      return jsonResponse_({ success: false, error: 'Wrong password' });
    }
    return jsonResponse_({
      success: true,
      role: role,
      bookings: listActiveBookings_(),
      seasons: listSeasons_(),
      cars: listCars_(),
      settings: listSettings_(),
      activityLog: listActivityLog_(),
      // Internal per-unit fleet registry (nickname/plate/VIN/status/notes) —
      // admin-only, never exposed via the public doGet responses below.
      fleetUnits: listFleetUnits_(),
    });
  }

  if (action === 'seasons') {
    return jsonResponse_({ seasons: listSeasons_() });
  }

  if (action === 'cars') {
    // Public: the live site's fleet section reads the car catalog from here.
    return jsonResponse_({ cars: listCars_() });
  }

  if (action === 'settings') {
    // Public: index.html applies SEO meta overrides (title/description) from here.
    return jsonResponse_({ settings: listSettings_() });
  }

  // Customer-initiated: the "My Account" bookings list. Auth is just
  // email+password re-checked on every call (see customerLogin above) —
  // there's no separate session-token layer, matching how admin.html
  // re-sends the admin password on every request.
  if (action === 'customerBookings') {
    const customer = findCustomerByEmail_(String(e.parameter.email || '').trim().toLowerCase());
    if (!customer || hashPassword_(e.parameter.password, customer.salt) !== customer.passwordHash) {
      return jsonResponse_({ success: false, error: 'invalid_credentials' });
    }
    const email = customer.email;
    const bookings = listActiveBookings_().filter(b => (b.email || '').trim().toLowerCase() === email);
    return jsonResponse_({ success: true, customer: { email: customer.email, name: customer.name, phone: customer.phone }, bookings: bookings });
  }

  // Public: for each car, how many units are currently booked overall (kept
  // for backward compatibility), every active booking's return date
  // (bookedReturns — the Nth soonest return date is the moment the (N+1)th
  // unit becomes free, used by the "notify me" feature), AND the full
  // pickup/return interval of each active booking (bookedIntervals) so the
  // site can tell whether a car is free for a SPECIFIC requested date range
  // rather than just "is it booked right now overall" — a car with a
  // booking next week must still be bookable for a period that doesn't
  // overlap it.
  const bookedCounts = {};
  const bookedReturns = {};
  const bookedIntervals = {};
  listActiveBookings_().forEach(b => {
    bookedCounts[b.carId] = (bookedCounts[b.carId] || 0) + 1;
    if (!bookedReturns[b.carId]) bookedReturns[b.carId] = [];
    if (b.ret) bookedReturns[b.carId].push(new Date(b.ret).toISOString());
    if (!bookedIntervals[b.carId]) bookedIntervals[b.carId] = [];
    if (b.pickup && b.ret) {
      bookedIntervals[b.carId].push({
        pickup: new Date(b.pickup).toISOString(),
        ret: new Date(b.ret).toISOString(),
        // Opaque unit id this specific booking is assigned to (blank for
        // legacy bookings made before the car had units registered) — needed
        // client-side to run the exact same per-unit algorithm the admin
        // dashboard uses (see getRemainingStockForRange in index.html and
        // findAvailableUnitForBooking_ above); never a plate/nickname, so
        // this stays safe to expose publicly.
        unitId: b.unitId || '',
      });
    }
  });
  Object.keys(bookedReturns).forEach(carId => bookedReturns[carId].sort());
  return jsonResponse_({
    bookedCounts: bookedCounts, bookedReturns: bookedReturns, bookedIntervals: bookedIntervals,
    // Per-car count of registered physical units (no plates/names — just a
    // number) — lets the site use the REAL unit count as the availability
    // denominator instead of the Cars sheet's `stock` field whenever the
    // owner has set up specific units for that car, keeping the calendar
    // shading customers see consistent with what booking creation enforces.
    carUnitCounts: countUnitsByCar_(),
    // The actual opaque unit ids per car (see listUnitIdsByCar_ above) — lets
    // the public site match each booking's unitId against the car's real
    // registered units, exactly like the admin dashboard and
    // findAvailableUnitForBooking_ do, instead of a bare count-vs-stock check.
    carUnitIds: listUnitIdsByCar_(),
  });
}

// Mandatory turnaround time between one booking's return and the next
// booking's pickup for the SAME physical car (cleaning/inspection buffer) —
// a car becomes bookable again exactly this many hours after its scheduled
// return, on the same day if that's when the buffer elapses.
const CAR_TURNAROUND_BUFFER_MS = 2 * 60 * 60 * 1000;

/* True if [startA, endA) conflicts with [startB, endB) once a mandatory
   CAR_TURNAROUND_BUFFER_MS gap is required between one range's end and the
   other's start (in either direction) — not just a bare interval overlap.
   Two bookings for the same car conflict unless one ends at least the buffer
   before the other starts; a booking ending at 10:00 does NOT conflict with
   one starting at 12:00 (exactly the 2h buffer later), but does conflict
   with one starting at 11:59. */
function rangesOverlap_(startA, endA, startB, endB) {
  return new Date(startB).getTime() < new Date(endA).getTime() + CAR_TURNAROUND_BUFFER_MS &&
         new Date(startA).getTime() < new Date(endB).getTime() + CAR_TURNAROUND_BUFFER_MS;
}

function doPost(e) {
  const body = JSON.parse(e.postData.contents);

  if (body.action === 'cancel') {
    const role = roleForPassword_(body.password);
    if (!role) return jsonResponse_({ success: false, error: 'Wrong password' });
    if (!body.row) return jsonResponse_({ success: false, error: 'Missing row' });
    getBookingsSheet_().deleteRow(Number(body.row));
    logActivity_(role, 'cancel_booking', 'Row ' + body.row);
    return jsonResponse_({ success: true });
  }

  // Admin-only: full edit of an existing reservation (website or manual) —
  // previously the dashboard only offered Cancel or a payment-status change,
  // with no way to open a booking and correct its dates or customer details.
  // Re-runs the same per-unit/stock availability check used when a booking is
  // first created, but EXCLUDING this booking's own row from that check (so a
  // booking editing its own unchanged dates doesn't look like a conflict with
  // itself), and re-assigns whichever unit is free for the new dates.
  if (body.action === 'editBooking') {
    const role = roleForPassword_(body.password);
    if (!role) return jsonResponse_({ success: false, error: 'Wrong password' });
    if (!body.row) return jsonResponse_({ success: false, error: 'Missing row' });
    const sheet = getBookingsSheet_();
    const rowNum = Number(body.row);
    const existing = sheet.getRange(rowNum, 1, 1, 19).getValues()[0];
    if (!existing[0]) return jsonResponse_({ success: false, error: 'Booking not found' });
    const carId = body.carId || existing[0];
    const stock = Number(body.stock) || Number(existing[8]) || 1;
    const pickup = body.pickup || existing[4];
    const ret = body.ret || existing[5];
    // body.unitId here is the admin's manual unit choice from the Edit
    // Reservation modal — role was already verified above to reach this
    // point, so it's safe to honor (re-validated against the live free list
    // inside findAvailableUnitForBooking_).
    const assignment = findAvailableUnitForBooking_(carId, pickup, ret, stock, rowNum, body.unitId || null);
    if (assignment.mode === 'sold_out') {
      return jsonResponse_({ success: false, error: 'sold_out' });
    }
    const phone = body.phone != null ? body.phone : existing[3];
    sheet.getRange(rowNum, 1, 1, 19).setValues([[
      carId,
      body.carName || existing[1],
      body.customerName != null ? body.customerName : existing[2],
      phone,
      pickup,
      ret,
      existing[6], // reference — never changes
      existing[7], // booked_at — never changes
      stock,
      body.total != null ? body.total : existing[9],
      existing[10], // source — never changes
      body.email != null ? body.email : existing[11],
      existing[12], existing[13], existing[14], existing[15], existing[16], // payment/document fields untouched
      body.nationality != null ? body.nationality : existing[17],
      assignment.unitId || '',
    ]]);
    sheet.getRange(rowNum, 4).setNumberFormat('@').setValue(phone);
    logActivity_(role, 'edit_booking', 'Row ' + rowNum + ' — ' + (body.carName || existing[1]));
    return jsonResponse_({ success: true, unitId: assignment.unitId || '', unitPlate: assignment.plate || '' });
  }

  if (body.action === 'addSeason') {
    const role = roleForPassword_(body.password);
    if (!role || EDITOR_ROLES.indexOf(role) === -1) return jsonResponse_({ success: false, error: 'Not allowed' });
    if (!body.name || !body.start || !body.end || !body.value) {
      return jsonResponse_({ success: false, error: 'Missing fields' });
    }
    getSeasonsSheet_().appendRow([
      body.name, body.start, body.end, body.type === 'fixed' ? 'fixed' : 'percent', Number(body.value),
    ]);
    logActivity_(role, 'add_season', body.name);
    return jsonResponse_({ success: true });
  }

  if (body.action === 'deleteSeason') {
    const role = roleForPassword_(body.password);
    if (!role || EDITOR_ROLES.indexOf(role) === -1) return jsonResponse_({ success: false, error: 'Not allowed' });
    if (!body.row) return jsonResponse_({ success: false, error: 'Missing row' });
    getSeasonsSheet_().deleteRow(Number(body.row));
    logActivity_(role, 'delete_season', 'Row ' + body.row);
    return jsonResponse_({ success: true });
  }

  if (body.action === 'saveCar') {
    const role = roleForPassword_(body.password);
    if (!role || EDITOR_ROLES.indexOf(role) === -1) return jsonResponse_({ success: false, error: 'Not allowed' });
    const car = body.car || {};
    if (!car.id || !car.name) return jsonResponse_({ success: false, error: 'Missing id/name' });
    upsertCar_(car);
    logActivity_(role, 'save_car', car.id + ' — ' + car.name);
    return jsonResponse_({ success: true });
  }

  if (body.action === 'deleteCar') {
    const role = roleForPassword_(body.password);
    if (!role || EDITOR_ROLES.indexOf(role) === -1) return jsonResponse_({ success: false, error: 'Not allowed' });
    if (!body.id) return jsonResponse_({ success: false, error: 'Missing id' });
    deleteCar_(body.id);
    logActivity_(role, 'delete_car', body.id);
    return jsonResponse_({ success: true });
  }

  // Admin-only: internal per-unit fleet registry (nickname, plate, fleet id,
  // VIN, status, notes) — lets the owner tell physically identical cars of
  // the same model apart (e.g. 5 units of "Clio 5 Automatique") without any
  // of this ever reaching the public site. Upserts by unitId; a blank/missing
  // unitId creates a new unit.
  if (body.action === 'saveFleetUnit') {
    const role = roleForPassword_(body.password);
    if (!role || EDITOR_ROLES.indexOf(role) === -1) return jsonResponse_({ success: false, error: 'Not allowed' });
    const unit = body.unit || {};
    if (!unit.carId) return jsonResponse_({ success: false, error: 'Missing carId' });
    const unitId = upsertFleetUnit_(unit);
    logActivity_(role, 'save_fleet_unit', unitId + ' — ' + (unit.nickname || unit.carId));
    return jsonResponse_({ success: true, unitId: unitId });
  }

  if (body.action === 'deleteFleetUnit') {
    const role = roleForPassword_(body.password);
    if (!role || EDITOR_ROLES.indexOf(role) === -1) return jsonResponse_({ success: false, error: 'Not allowed' });
    if (!body.unitId) return jsonResponse_({ success: false, error: 'Missing unitId' });
    deleteFleetUnit_(body.unitId);
    logActivity_(role, 'delete_fleet_unit', body.unitId);
    return jsonResponse_({ success: true });
  }

  if (body.action === 'duplicateCar') {
    const role = roleForPassword_(body.password);
    if (!role || EDITOR_ROLES.indexOf(role) === -1) return jsonResponse_({ success: false, error: 'Not allowed' });
    if (!body.id) return jsonResponse_({ success: false, error: 'Missing id' });
    const newId = duplicateCar_(body.id);
    if (!newId) return jsonResponse_({ success: false, error: 'Car not found' });
    logActivity_(role, 'duplicate_car', body.id + ' -> ' + newId);
    return jsonResponse_({ success: true, id: newId });
  }

  if (body.action === 'saveSetting') {
    const role = roleForPassword_(body.password);
    if (!role || EDITOR_ROLES.indexOf(role) === -1) return jsonResponse_({ success: false, error: 'Not allowed' });
    if (!body.key) return jsonResponse_({ success: false, error: 'Missing key' });
    setSetting_(body.key, body.value || '');
    logActivity_(role, 'save_setting', body.key);
    return jsonResponse_({ success: true });
  }

  // Customer-initiated: called right after they pick a payment method in the
  // payment step (Bank Transfer "I've paid" click, or Wise "Pay with Wise"
  // click). No password — the customer isn't authenticated at this point.
  if (body.action === 'setBookingPayment') {
    if (!body.ref) return jsonResponse_({ success: false, error: 'Missing ref' });
    const row = findBookingRowByRef_(body.ref);
    if (row === -1) return jsonResponse_({ success: false, error: 'Booking not found' });
    const sheet = getBookingsSheet_();
    sheet.getRange(row, 13, 1, 3).setValues([[
      body.paymentMethod || 'none',
      body.paymentStatus || 'pending',
      body.bookingStatus || 'pending_payment',
    ]]);
    return jsonResponse_({ success: true });
  }

  // Customer-initiated: called from the browser in the background right after
  // a booking is recorded, to upload the customer's driving license / passport
  // photo to Drive and attach the resulting link to that booking row. This is
  // what lets the confirmation PDF's "Driving license" / "Passport" bullets be
  // clickable links to the actual photo, instead of just showing the filename
  // the customer picked. No password — the customer isn't authenticated here.
  if (body.action === 'attachBookingDocument') {
    if (!body.ref || !body.docType || !body.dataBase64) return jsonResponse_({ success: false, error: 'Missing fields' });
    if (body.docType !== 'license' && body.docType !== 'passport') return jsonResponse_({ success: false, error: 'Invalid docType' });
    const row = findBookingRowByRef_(body.ref);
    if (row === -1) return jsonResponse_({ success: false, error: 'Booking not found' });
    try {
      const bytes = Utilities.base64Decode(body.dataBase64);
      const mimeType = body.mimeType || 'application/octet-stream';
      const safeName = (body.docType + '_' + body.ref + '_' + (body.fileName || 'document')).replace(/[\/\\]/g, '_');
      const blob = Utilities.newBlob(bytes, mimeType, safeName);
      const file = getBookingDocumentsFolder_().createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      const url = 'https://drive.google.com/file/d/' + file.getId() + '/view';
      const sheet = getBookingsSheet_();
      sheet.getRange(row, body.docType === 'license' ? 16 : 17).setValue(url);
      return jsonResponse_({ success: true, url: url });
    } catch (err) {
      return jsonResponse_({ success: false, error: 'upload_failed', message: String(err && err.message || err) });
    }
  }

  // Admin-only: change a booking's payment status from the dashboard
  // (Pending / Awaiting Verification / Paid / Cancelled / Refunded).
  // Marking it Paid auto-confirms the booking and emails the customer.
  if (body.action === 'adminSetPaymentStatus') {
    const role = roleForPassword_(body.password);
    if (!role) return jsonResponse_({ success: false, error: 'Wrong password' });
    if (!body.ref || !body.paymentStatus) return jsonResponse_({ success: false, error: 'Missing fields' });
    const row = findBookingRowByRef_(body.ref);
    if (row === -1) return jsonResponse_({ success: false, error: 'Booking not found' });
    const sheet = getBookingsSheet_();
    sheet.getRange(row, 14).setValue(body.paymentStatus);
    let bookingStatus = sheet.getRange(row, 15).getValue() || 'pending_payment';
    if (body.paymentStatus === 'paid') {
      bookingStatus = 'confirmed';
      sheet.getRange(row, 15).setValue(bookingStatus);
      sendPaymentConfirmedEmail_(sheet.getRange(row, 1, 1, 15).getValues()[0]);
    } else if (body.paymentStatus === 'cancelled') {
      bookingStatus = 'cancelled';
      sheet.getRange(row, 15).setValue(bookingStatus);
    }
    logActivity_(role, 'set_payment_status', body.ref + ' — ' + body.paymentStatus);
    return jsonResponse_({ success: true });
  }

  // Customer-initiated: a visitor asked to be told when a sold-out car frees
  // up. No password — stored for autoExpireBookings (see below) to pick up
  // and auto-email once the car has free stock again.
  if (body.action === 'requestAvailabilityNotification') {
    if (!body.carId || !body.contact) return jsonResponse_({ success: false, error: 'Missing fields' });
    getAvailabilityRequestsSheet_().appendRow([
      body.carId, body.carName || '', body.contact, new Date(), '',
    ]);
    return jsonResponse_({ success: true });
  }

  // ---- Customer accounts (self-service signup/login/profile) ----
  // Customer-initiated: create an account, normally right after a booking
  // (see index.html's "create an account" checkbox). No admin password
  // involved — this is a completely separate, low-privilege account system
  // from the ADMIN_ROLES used by the dashboard.
  if (body.action === 'customerSignup') {
    if (!body.email || !body.password) return jsonResponse_({ success: false, error: 'Missing fields' });
    const email = String(body.email).trim().toLowerCase();
    if (!email || email.indexOf('@') === -1) return jsonResponse_({ success: false, error: 'invalid_email' });
    if (String(body.password).length < 6) return jsonResponse_({ success: false, error: 'weak_password' });
    if (findCustomerByEmail_(email)) return jsonResponse_({ success: false, error: 'email_taken' });
    const salt = generateSalt_();
    getCustomersSheet_().appendRow([
      email, hashPassword_(body.password, salt), salt,
      body.name || '', body.phone || '', new Date(),
    ]);
    return jsonResponse_({ success: true, customer: { email: email, name: body.name || '', phone: body.phone || '' } });
  }

  // Customer-initiated: log in with email/password. Returns the customer's
  // own profile fields on success — the frontend then stores {email,
  // password} in localStorage (mirroring how admin.html keeps the admin
  // password in sessionStorage) and resends both on every subsequent
  // customerBookings/customerUpdateProfile call, since there's no separate
  // session-token layer here.
  if (body.action === 'customerLogin') {
    if (!body.email || !body.password) return jsonResponse_({ success: false, error: 'Missing fields' });
    const customer = findCustomerByEmail_(String(body.email).trim().toLowerCase());
    if (!customer || hashPassword_(body.password, customer.salt) !== customer.passwordHash) {
      return jsonResponse_({ success: false, error: 'invalid_credentials' });
    }
    return jsonResponse_({ success: true, customer: { email: customer.email, name: customer.name, phone: customer.phone } });
  }

  // Customer-initiated: update their own name/phone. Re-verifies the
  // password (same as every other customer action) before writing.
  if (body.action === 'customerUpdateProfile') {
    if (!body.email || !body.password) return jsonResponse_({ success: false, error: 'Missing fields' });
    const customer = findCustomerByEmail_(String(body.email).trim().toLowerCase());
    if (!customer || hashPassword_(body.password, customer.salt) !== customer.passwordHash) {
      return jsonResponse_({ success: false, error: 'invalid_credentials' });
    }
    getCustomersSheet_().getRange(customer.row, 4, 1, 2).setValues([[
      body.name != null ? body.name : customer.name,
      body.phone != null ? body.phone : customer.phone,
    ]]);
    return jsonResponse_({ success: true });
  }

  // Customer-initiated: cancel one of their own reservations from the "My
  // Account" dashboard. Ownership is checked against the booking's stored
  // customer_email (not just "any authenticated customer") so one customer
  // can never cancel another's reservation by guessing a reference.
  if (body.action === 'customerCancelBooking') {
    if (!body.email || !body.password || !body.ref) return jsonResponse_({ success: false, error: 'Missing fields' });
    const customer = findCustomerByEmail_(String(body.email).trim().toLowerCase());
    if (!customer || hashPassword_(body.password, customer.salt) !== customer.passwordHash) {
      return jsonResponse_({ success: false, error: 'invalid_credentials' });
    }
    const row = findBookingRowByRef_(body.ref);
    if (row === -1) return jsonResponse_({ success: false, error: 'Booking not found' });
    const sheet = getBookingsSheet_();
    const ownerEmail = String(sheet.getRange(row, 12).getValue() || '').trim().toLowerCase();
    if (ownerEmail !== customer.email.trim().toLowerCase()) return jsonResponse_({ success: false, error: 'not_owner' });
    sheet.deleteRow(row);
    return jsonResponse_({ success: true });
  }

  // Customer-initiated: change their own reservation's pickup/return dates
  // from the "My Account" dashboard. Everything else about the booking
  // (vehicle, price, customer details) stays untouched — this is a
  // self-service date change, not a full edit. Re-runs the same
  // availability check used everywhere else (excluding this booking's own
  // row) so a customer can never double-book a unit that's since been
  // taken by someone else.
  if (body.action === 'customerEditBooking') {
    if (!body.email || !body.password || !body.ref || !body.pickup || !body.ret) {
      return jsonResponse_({ success: false, error: 'Missing fields' });
    }
    const customer = findCustomerByEmail_(String(body.email).trim().toLowerCase());
    if (!customer || hashPassword_(body.password, customer.salt) !== customer.passwordHash) {
      return jsonResponse_({ success: false, error: 'invalid_credentials' });
    }
    const row = findBookingRowByRef_(body.ref);
    if (row === -1) return jsonResponse_({ success: false, error: 'Booking not found' });
    const sheet = getBookingsSheet_();
    const existing = sheet.getRange(row, 1, 1, 19).getValues()[0];
    const ownerEmail = String(existing[11] || '').trim().toLowerCase();
    if (ownerEmail !== customer.email.trim().toLowerCase()) return jsonResponse_({ success: false, error: 'not_owner' });
    const carId = existing[0];
    const stock = Number(existing[8]) || 1;
    const assignment = findAvailableUnitForBooking_(carId, body.pickup, body.ret, stock, row, null);
    if (assignment.mode === 'sold_out') return jsonResponse_({ success: false, error: 'sold_out' });
    sheet.getRange(row, 5, 1, 2).setValues([[body.pickup, body.ret]]);
    sheet.getRange(row, 19).setValue(assignment.unitId || '');
    return jsonResponse_({ success: true });
  }

  // Default action: record a new booking (website checkout OR a manual
  // reservation created from the admin dashboard — same code path, so
  // inventory updates automatically either way).
  if (!body.carId) {
    return jsonResponse_({ success: false, error: 'Missing carId' });
  }
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const stock = Number(body.stock) || 1;
    // adminPassword is only ever sent by the admin dashboard's manual
    // reservation form (see admin.html), never by the public website
    // checkout — so a client-supplied unitId (the admin's manual choice of
    // which physical unit to assign, from the Reservations > + Manual
    // reservation modal) is only honored once we've confirmed it actually
    // came from an authenticated admin, never from an anonymous customer.
    const manualRole = body.adminPassword ? roleForPassword_(body.adminPassword) : null;
    const manualUnitId = (body.unitId && manualRole) ? body.unitId : null;
    // Per-unit assignment: if this car has specific units registered (plate/
    // fleet ID, entered by the owner in the admin "Fleet Units" screen), the
    // booking is checked and assigned against those REAL physical units —
    // this is what actually prevents "all units of this model are booked" from
    // ever showing a free date before every one of them frees up, since it's
    // no longer just a bare count. Cars with no units registered yet fall back
    // to the previous count-vs-stock check unchanged.
    const assignment = findAvailableUnitForBooking_(body.carId, body.pickup, body.ret, stock, null, manualUnitId);
    if (assignment.mode === 'sold_out') {
      return jsonResponse_({ success: false, error: 'sold_out' });
    }
    const bookingsSheet = getBookingsSheet_();
    bookingsSheet.appendRow([
      body.carId,
      body.carName || '',
      body.customerName || '',
      body.phone || '',
      body.pickup || '',
      body.ret || '',
      body.ref || '',
      new Date(),
      stock,
      body.total || '',
      body.source || 'website',
      body.email || '',
      'none',
      'pending',
      'pending_payment',
      '',
      '',
      body.nationality || '',
      assignment.unitId || '',
    ]);
    // Phone numbers start with "+" (country code, e.g. "+212 6..."), which
    // Google Sheets auto-parses as the start of a formula/number and shows
    // as "#ERROR!" — force that cell to plain text and rewrite it so it
    // displays exactly as entered.
    bookingsSheet.getRange(bookingsSheet.getLastRow(), 4).setNumberFormat('@').setValue(body.phone || '');
    sendBookingEmail_(body);
    if (manualRole) logActivity_(manualRole, 'manual_booking', (body.source || 'manual') + ' — ' + body.carName);
    return jsonResponse_({ success: true, unitId: assignment.unitId || '', unitPlate: assignment.plate || '' });
  } finally {
    lock.releaseLock();
  }
}

/* Decides which physical unit (if any) a new booking should be assigned to.
   - If the car has no units registered in the FleetUnits sheet yet, falls
     back to the previous behavior: just count overlapping bookings against
     the car's abstract `stock` number (no specific unit is assigned).
   - If units ARE registered, availability becomes exact: a day is only ever
     truly sold out once every registered unit has an overlapping booking
     assigned to it — this is what makes "extend one unit's booking to
     September" correctly leave the OTHER unit (and only that unit) bookable,
     instead of a bare count that can't tell units apart.
   - manualUnitId: the admin dashboard's manual reservation/edit modal lets
     the owner pick exactly which free unit to assign when more than one is
     available, instead of always silently auto-assigning — so the agency can
     balance bookings across its fleet by its own plan. Only call sites that
     already verified an authenticated admin role pass this through (see
     doPost); it is re-validated here against the actual free list so a stale
     / raced choice can never double-book a unit. */
function findAvailableUnitForBooking_(carId, pickup, ret, stock, excludeRow, manualUnitId) {
  const units = listFleetUnits_().filter(u => u.carId === carId);
  const bookingsPool = excludeRow ? listActiveBookings_().filter(b => b.row !== excludeRow) : listActiveBookings_();
  if (units.length === 0) {
    const overlapping = pickup && ret
      ? bookingsPool.filter(b => b.carId === carId && rangesOverlap_(pickup, ret, b.pickup, b.ret)).length
      : bookingsPool.filter(b => b.carId === carId).length;
    return overlapping >= stock ? { mode: 'sold_out' } : { mode: 'legacy' };
  }
  const overlappingBookings = pickup && ret
    ? bookingsPool.filter(b => b.carId === carId && rangesOverlap_(pickup, ret, b.pickup, b.ret))
    : bookingsPool.filter(b => b.carId === carId);
  const assignedOverlapUnitIds = overlappingBookings.filter(b => b.unitId).map(b => b.unitId);
  // Bookings made before a car had units registered (or any other case where
  // a booking has no unitId) can't be matched to a specific unit — but they
  // still occupy a real physical car, so each one must still claim one of the
  // free slots below. Otherwise a car with older, unassigned bookings would
  // wrongly look fully free just because none of them "match" a unit ID.
  const legacyOverlapCount = overlappingBookings.filter(b => !b.unitId).length;
  const freeUnits = units.filter(u => assignedOverlapUnitIds.indexOf(u.unitId) === -1);
  if (manualUnitId) {
    const requested = freeUnits.find(u => u.unitId === manualUnitId);
    if (!requested) return { mode: 'sold_out' };
    return { mode: 'unit', unitId: requested.unitId, plate: requested.plate || requested.nickname || requested.unitId };
  }
  if (freeUnits.length <= legacyOverlapCount) return { mode: 'sold_out' };
  const free = freeUnits[legacyOverlapCount];
  return { mode: 'unit', unitId: free.unitId, plate: free.plate || free.nickname || free.unitId };
}

/* Frees up stock automatically once a booking's return date has passed.
   Installed as an hourly time-driven trigger (see v3 setup notes). Also
   checks pending "notify me" requests afterwards, since freeing stock here
   is exactly the moment a sold-out car might become bookable again. */
function autoExpireBookings() {
  const sheet = getBookingsSheet_();
  const rows = sheet.getDataRange().getValues();
  const now = new Date();
  for (let i = rows.length - 1; i >= 1; i--) {
    const ret = rows[i][5];
    if (ret && new Date(ret) < now) {
      sheet.deleteRow(i + 1);
    }
  }
  notifyAvailabilityRequests_();
}

/* For each pending "notify me when available" request, checks whether that
   car currently has free stock; if so, emails the requester (when their
   contact looks like an email — no SMS integration for phone numbers) and
   removes the request so it isn't processed again. Phone-only requests are
   just removed once available, for the team to follow up manually. */
function notifyAvailabilityRequests_() {
  const sheet = getAvailabilityRequestsSheet_();
  const rows = sheet.getDataRange().getValues();
  if (rows.length < 2) return;
  const cars = listCars_();
  const bookedCounts = {};
  listActiveBookings_().forEach(b => { bookedCounts[b.carId] = (bookedCounts[b.carId] || 0) + 1; });
  for (let i = rows.length - 1; i >= 1; i--) {
    const carId = rows[i][0];
    const car = cars.find(c => c.id === carId);
    if (!car) continue;
    const remaining = (car.stock || 1) - (bookedCounts[carId] || 0);
    if (remaining <= 0) continue;
    const contact = rows[i][2];
    if (contact && contact.indexOf('@') !== -1) {
      try {
        MailApp.sendEmail(
          contact,
          'Good news — ' + (rows[i][1] || carId) + ' is available again!',
          'The car you asked about, ' + (rows[i][1] || carId) + ', is now available for booking.\n\nBook it here before it\'s gone: ' + (SITE_URL || '')
        );
      } catch (e) { /* don't fail the whole batch over one email quota/error */ }
    }
    sheet.deleteRow(i + 1);
  }
}

/* ---- Cars (fleet catalog) ---- */
function listCars_() {
  const rows = getCarsSheet_().getDataRange().getValues();
  const cars = [];
  for (let i = 1; i < rows.length; i++) {
    if (!rows[i][0]) continue;
    cars.push({
      id: rows[i][0],
      name: rows[i][1],
      price: Number(rows[i][2]) || 0,
      seats: Number(rows[i][3]) || 5,
      gearKey: rows[i][4] || 'manual',
      doors: Number(rows[i][5]) || 4,
      fuelKey: rows[i][6] || 'diesel',
      body: rows[i][7] ? String(rows[i][7]).split(',').map(s => s.trim()).filter(Boolean) : [],
      cat: rows[i][8] || 'citadine',
      stock: Number(rows[i][9]) || 1,
      imgSrc: rows[i][10] || '',
      offer: rows[i][11] || (rows[i][2] + '€'),
      sortOrder: rows[i][12] === '' || rows[i][12] == null ? i : Number(rows[i][12]),
    });
  }
  cars.sort((a, b) => a.sortOrder - b.sortOrder);
  return cars;
}

function upsertCar_(car) {
  const sheet = getCarsSheet_();
  const rows = sheet.getDataRange().getValues();
  const rowData = [
    car.id, car.name, Number(car.price) || 0, Number(car.seats) || 5, car.gearKey || 'manual',
    Number(car.doors) || 4, car.fuelKey || 'diesel', Array.isArray(car.body) ? car.body.join(',') : (car.body || ''),
    car.cat || 'citadine', Number(car.stock) || 1, car.imgSrc || '', car.offer || ((Number(car.price) || 0) + '€'),
    car.sortOrder == null ? '' : Number(car.sortOrder),
  ];
  for (let i = 1; i < rows.length; i++) {
    if (rows[i][0] === car.id) {
      sheet.getRange(i + 1, 1, 1, rowData.length).setValues([rowData]);
      return;
    }
  }
  sheet.appendRow(rowData);
}

function deleteCar_(id) {
  const sheet = getCarsSheet_();
  const rows = sheet.getDataRange().getValues();
  for (let i = rows.length - 1; i >= 1; i--) {
    if (rows[i][0] === id) {
      sheet.deleteRow(i + 1);
      return;
    }
  }
}

function duplicateCar_(id) {
  const rows = getCarsSheet_().getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (rows[i][0] === id) {
      const newId = id + '-copy-' + Date.now().toString().slice(-6);
      const rowData = rows[i].slice();
      rowData[0] = newId;
      rowData[1] = rowData[1] + ' (Copy)';
      getCarsSheet_().appendRow(rowData);
      return newId;
    }
  }
  return null;
}

/* Same idea as countUnitsByCar_, but the actual opaque unitId strings per car
   instead of just a count — safe to expose publicly (a bare "FU-XXXXX" id
   reveals no plate/nickname/status), and needed so index.html can run the
   EXACT same per-unit availability algorithm the admin dashboard and
   findAvailableUnitForBooking_ use (matching a booking's unitId against the
   car's actual registered units), instead of a simpler count-only check that
   could disagree with the admin panel in edge cases involving legacy
   (no-unitId) bookings mixed with unit-assigned ones. */

  });
  return ids;
}

/* ---- Customer accounts ----
   A separate, low-privilege account system from ADMIN_ROLES: lets a
   customer sign up with email/password during booking, log back in on a
   future visit, and see only their own bookings (matched by email against
   the Bookings sheet's existing customer_email column — no schema change
   needed there). Passwords are never stored in plain text: each account
   gets its own random salt, and only a salted SHA-256 hash is saved. */
function getCustomersSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(CUSTOMERS_SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(CUSTOMERS_SHEET_NAME);
    sheet.appendRow(['email', 'password_hash', 'salt', 'name', 'phone', 'created_at']);
  }
  return sheet;
}
function generateSalt_() {
  return Utilities.getUuid();
}
function hashPassword_(password, salt) {
  const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(password) + salt);
  return digest.map(b => (b < 0 ? b + 256 : b).toString(16).padStart(2, '0')).join('');
}
/* Finds a customer account by email (case-insensitive — caller is expected
   to have already lowercased/trimmed it). Returns null if not found. */
function findCustomerByEmail_(email) {
  if (!email) return null;
  const rows = getCustomersSheet_().getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]).trim().toLowerCase() === email) {
      return {
        row: i + 1,
        email: rows[i][0],
        passwordHash: rows[i][1],
        salt: rows[i][2],
        name: rows[i][3] || '',
        phone: rows[i][4] || '',
        createdAt: rows[i][5],
      };
    }
  }
  return null;
}
